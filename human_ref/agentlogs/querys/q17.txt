SELECT
  trace_id,
  observation_id,
  seq_no,
  type,
  status,
  json_extract_string(payload, '$.attr.request_key') AS request_key,
  json_extract_string(payload, '$.attr.workflow_variant') AS workflow_variant
FROM agent_observations
WHERE tenant = '[TENANT]'
  AND json_extract_string(payload, '$.attr.request_key') = '[REQUEST_KEY]'
  AND json_extract_string(payload, '$.attr.workflow_variant') = '[WORKFLOW_VARIANT]'
ORDER BY seq_no ASC, observation_id ASC
LIMIT 100;
